cmake_minimum_required(VERSION 3.16)
project(tinyrt_runtime_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/HostWamr.cmake")
add_executable(test_runtime test_runtime.c)
target_include_directories(test_runtime PRIVATE ../../runtime/include "${WAMR_ROOT_DIR}/core/iwasm/include")
target_compile_definitions(test_runtime PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_BINARY_DIR}/fixtures" EXAMPLE_DIR="${CMAKE_CURRENT_LIST_DIR}/../fixtures/guests")
target_link_libraries(test_runtime PRIVATE tinyrt_runtime ws2_32)
if(MSVC)
 target_compile_options(test_runtime PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
 "${CMAKE_CURRENT_LIST_DIR}/make_fixtures.py"
 "${CMAKE_CURRENT_LIST_DIR}/make_round_fixtures.py"
 "${CMAKE_CURRENT_LIST_DIR}/make_pixel_fixtures.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/make_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" COMMAND_ERROR_IS_FATAL ANY)
enable_testing()
add_test(NAME runtime COMMAND test_runtime)
set_tests_properties(runtime PROPERTIES TIMEOUT 30)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/make_round_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" COMMAND_ERROR_IS_FATAL ANY)
add_executable(test_round test_round.c)
target_link_libraries(test_round PRIVATE tinyrt_runtime)
target_compile_definitions(test_round PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_BINARY_DIR}/fixtures")
if(MSVC)
 target_compile_options(test_round PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
add_test(NAME round_stop COMMAND test_round)
set_tests_properties(round_stop PROPERTIES TIMEOUT 30)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/make_pixel_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" COMMAND_ERROR_IS_FATAL ANY)
add_executable(test_pixels test_pixels.c)
target_link_libraries(test_pixels PRIVATE tinyrt_runtime)
target_compile_definitions(test_pixels PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_BINARY_DIR}/fixtures")
if(MSVC)
 target_compile_options(test_pixels PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
add_test(NAME pixels_clock COMMAND test_pixels)
set_tests_properties(pixels_clock PROPERTIES TIMEOUT 30)
