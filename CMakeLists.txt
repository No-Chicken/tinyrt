cmake_minimum_required(VERSION 3.24)
project(tinyrt_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT WIN32)
  message(FATAL_ERROR "Full core host tests currently require Windows BCrypt; Linux has not been validated.")
endif()
add_test(NAME contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts/test_contracts.py")
add_test(NAME frozen_guests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/verify_guests.py")
foreach(suite store package runtime manager integration transport)
  add_subdirectory("tests/${suite}")
endforeach()

add_executable(tinyrt-run tools/runtime_runner.c)
target_link_libraries(tinyrt-run PRIVATE tinyrt_runtime)
if(MSVC)
  target_compile_options(tinyrt-run PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
add_test(NAME runner COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/runner/test_runner.py")
set_tests_properties(runner PROPERTIES ENVIRONMENT "TINYRT_RUNNER=$<TARGET_FILE:tinyrt-run>;TINYRT_FIXTURE_DIR=${CMAKE_CURRENT_BINARY_DIR}/tests/runtime/fixtures")
