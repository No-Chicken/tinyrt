cmake_minimum_required(VERSION 3.16)
project(tinyrt_store_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(test_store test_store.c fake_nor.c test_package_verifier.c
    ../../runtime/store/tinyrt_store.c ../../runtime/store/tinyrt_directory.c)
target_include_directories(test_store PRIVATE ../../runtime/include ../../runtime/store)
if(MSVC)
    target_compile_options(test_store PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_store PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME store COMMAND test_store)
set_tests_properties(store PROPERTIES TIMEOUT 120)
