cmake_minimum_required(VERSION 3.16)
project(tinyrt_package_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(test_package test_package.c ../../runtime/package/package.c ../../runtime/package/package_bcrypt.c)
target_link_libraries(test_package PRIVATE bcrypt)
target_include_directories(test_package PRIVATE ../../runtime/include)
if(MSVC)
  target_compile_definitions(test_package PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(test_package PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(test_package PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME fixtures COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set_tests_properties(fixtures PROPERTIES FIXTURES_SETUP package_fixtures)
add_test(NAME package COMMAND test_package "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set_tests_properties(package PROPERTIES FIXTURES_REQUIRED package_fixtures)
