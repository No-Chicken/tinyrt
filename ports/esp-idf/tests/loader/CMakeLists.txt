cmake_minimum_required(VERSION 3.20)
project(tinyrt_loader_failure C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(WAMR_BUILD_PLATFORM windows)
set(WAMR_BUILD_TARGET X86_64)
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_LOOP_POLL 1)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
add_definitions(-DCOMPILING_WASM_RUNTIME_API=1)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
set_source_files_properties(${WAMR_ROOT_DIR}/core/shared/platform/windows/win_memmap.c
    PROPERTIES COMPILE_DEFINITIONS "os_mprotect=real_os_mprotect;os_mmap=real_os_mmap;os_munmap=real_os_munmap")
add_executable(loader_failure loader_failure.c ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(loader_failure PRIVATE ${WAMR_ROOT_DIR}/core/iwasm/include)
target_link_libraries(loader_failure ntdll)
target_compile_options(loader_failure PRIVATE /utf-8)
