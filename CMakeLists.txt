cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_LIST_DIR}/VERSION" TINYRT_VERSION)
string(STRIP "${TINYRT_VERSION}" TINYRT_VERSION)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/VERSION")
string(REGEX MATCH "^[0-9]+[.][0-9]+[.][0-9]+" TINYRT_CMAKE_VERSION "${TINYRT_VERSION}")
project(tinyrt_core VERSION "${TINYRT_CMAKE_VERSION}" LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts/test_contracts.py")
add_test(NAME frozen_guests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/verify_guests.py")
if(WIN32)
  foreach(suite store package runtime manager integration transport esp_guard)
    add_subdirectory("tests/${suite}")
  endforeach()
else()
  # The full package test harness uses BCrypt; the preview runner does not.
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/HostWamr.cmake")
endif()

add_library(tinyrt_mic_host STATIC runtime/audio/tinyrt_mic_features.c tools/mic_wav.c)
target_include_directories(tinyrt_mic_host PUBLIC runtime/audio)
if(NOT MSVC)
  target_link_libraries(tinyrt_mic_host PUBLIC m)
endif()
add_executable(tinyrt-mic-features tools/mic_features_cli.c)
target_link_libraries(tinyrt-mic-features PRIVATE tinyrt_mic_host)
add_test(NAME mic_features COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime/test_mic_features.py" $<TARGET_FILE:tinyrt-mic-features>)

add_executable(tinyrt-run tools/runtime_runner.c tools/mic_replay.c)
target_link_libraries(tinyrt-run PRIVATE tinyrt_runtime tinyrt_mic_host)
if(WIN32)
  target_link_libraries(tinyrt-run PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(tinyrt-run PRIVATE /utf-8 /W4 /WX /wd4996)
endif()
if(WIN32)
add_test(NAME runner COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/runner/test_runner.py")
set_tests_properties(runner PROPERTIES ENVIRONMENT "TINYRT_RUNNER=$<TARGET_FILE:tinyrt-run>;TINYRT_FIXTURE_DIR=${CMAKE_CURRENT_BINARY_DIR}/tests/runtime/fixtures")
endif()
