cmake_minimum_required(VERSION 3.16)
project(tinyrt_transport_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(test_core_transport test_transport.c ../../protocol/tinyrt_transport.c)
target_include_directories(test_core_transport PRIVATE ../../protocol)
if(MSVC)
  target_compile_options(test_core_transport PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(test_core_transport PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME core_transport COMMAND test_core_transport)
